二、玩家与属性系统
3. 玩家属性管理
// 玩家属性接口 (ISP: 分离不同属性关注点)
public interface IHealthSystem
{
    int CurrentHP { get; }
    int MaxHP { get; }
    void TakeDamage(int damage);
    void Heal(int amount);
}
public interface ISanitySystem  
{
    int CurrentSan { get; }
    int MaxSan { get; }
    void LoseSanity(int amount);
    void RestoreSanity(int amount);
}
// 玩家控制器 (SRP: 只负责玩家行为协调)
public class Player : MonoBehaviour, IHealthSystem, ISanitySystem
{
    // 属性系统委托给专门的组件
    [SerializeField] private HealthComponent healthComponent;
    [SerializeField] private SanityComponent sanityComponent;
    
    void Move(Vector2 direction)
    void Attack()
    void UseSkill(int skillIndex)
}​

七、敌人系统
11. 敌人框架
// 敌人接口 (ISP: 敌人行为分离)
public interface IEnemy : IDamageable, IHealable
{
    void Initialize(EnemyData data);
    void StartAI();
    void StopAI();
}
// 敌人基类 (Template Method + LSP)
public abstract class Enemy : MonoBehaviour, IEnemy
{
    protected EnemyData enemyData;
    protected StateMachine<Enemy> stateMachine;
    
    public abstract void Initialize(EnemyData data);
    public virtual void TakeDamage(int damage, DamageType type)
    public virtual void Heal(int amount)
}
// 具体敌人实现 (OCP: 新敌人类型)
public class BasicEnemy : Enemy
public class BossEnemy : Enemy
public class HealableEnemy : Enemy​
12. AI状态机
// 状态机组件 (Composition over Inheritance)
public class StateMachine<T> where T : class
{
    void ChangeState(IState<T> newState)
    void Update()
}
// AI状态接口
public interface IState<T>
{
    void Enter(T owner);
    void Update(T owner);
    void Exit(T owner);
}
// 可复用的AI状态 (RP: 状态可复用)
public class IdleState : IState<Enemy>
public class ChaseState : IState<Enemy>
public class AttackState : IState<Enemy>
public class FleeState : IState<Enemy>​
